    def make_card(variant, *, memory, chain_type):
    card = ExpCardInit()
    card.proj_dir      = RUN_DIR / variant
    card.projectname   = variant
    card.model         = MODEL_NAME
    card.family        = MODEL_FAMILY
    card.parameters    = {'temperature': 0}
    card.task_file     = TASK_FILE
    card.parser        = '0'   # free text ('match'/'no-match'), no structured schema
    card.cogtype       = 'no'
    card.nsim          = 1
    card.tunnel_status = '0'
    card.memory        = memory
    card.chain_type    = chain_type
    return card


def decode(pred_resp):
    return pred_resp.content if hasattr(pred_resp, 'content') else str(pred_resp)

    Trial-chain variant¶

    Same task, real conversational memory across trials instead of none: chain_type='trial', memory='Convo' — the n-back generator's own design note explains why this uses psychscanner's real trial-chain memory rather than a hand-baked history string in the prompt (see examples/tasks/generators/generate_nback_sweetpea.py). parser='0' — the card has no structured schema; grading below is a plain substring match against corrAns.

def load_judgments(csv_path):
    df = pd.read_csv(csv_path)
    def cat(s):
        s = str(s).lower()
        if 'no-match' in s or 'no match' in s:
            return 'no-match'
        if 'match' in s:
            return 'match'
        return 'unparsed'
    df['judged'] = df['pred_resp_raw'].apply(cat)
    return df

    N-back accuracy¶

    2-back accuracy for the two runs above (default temperature=0, nsim=1, no feedback) — the model's match/no-match judgment compared against each trial's corrAns. See 05_nback_task.ipynb for the same task run across all four memory/feedback configurations, including what happens when corrective feedback is added.

    import json as _json

_task = _json.loads(TASK_FILE.read_text())
_CORR_ANS = {trcode: trials[0]['corrAns'] for trcode, trials in _task['items'].items()}

def load_accuracy(csv_path):
    df = pd.read_csv(csv_path)
    def cat(s):
        s = str(s).lower()
        if 'no-match' in s or 'no match' in s:
            return 'no-match'
        if 'match' in s:
            return 'match'
        return None
    df['judged'] = df['pred_resp_raw'].apply(cat)
    df['corrAns'] = df['trcode'].map(_CORR_ANS)
    correct = (df['judged'] == df['corrAns']).sum()
    return correct, len(df)

correct_st, n_st = load_accuracy(OUT / 'nback_singleturn.csv')
correct_tc, n_tc = load_accuracy(OUT / 'nback_trialchain.csv')

fig, ax = plt.subplots(figsize=(5, 4))
labels = ['Single-Turn', 'Trial-Chain']
accs = [correct_st / n_st, correct_tc / n_tc]
bars = ax.bar(labels, accs, color=['steelblue', 'gray'], width=0.5)
for bar, acc, correct, n in zip(bars, accs, [correct_st, correct_tc], [n_st, n_tc]):
    ax.text(bar.get_x() + bar.get_width() / 2, acc + 0.03, f'{correct}/{n}', ha='center', fontsize=10)
ax.axhline(18/22, color='black', linestyle=':', linewidth=1, label='majority-class baseline (82%)')
ax.set_ylim(0, 1.15); ax.set_yticks([0, 0.25, 0.5, 0.75, 1.0])
ax.set_ylabel('2-back accuracy')
ax.set_title(f'N-back accuracy · model: {MODEL_NAME}')
ax.legend(fontsize=8, loc='upper right')
ax.grid(axis='y', linestyle='--', alpha=0.4)
ax.spines[['top', 'right']].set_visible(False)
plt.tight_layout()
fig.savefig(OUT / 'nback_accuracy_figure.png', dpi=120, bbox_inches='tight')

    df = pd.read_csv(csv_path)

    Further reading¶

    From the psychscanner team, on the two tasks run above:

    1. "Reality Monitoring in Large Language Models: Self-Knowledge That Transforms with Conversation Memory" (Ranjan, Sokratous & Odegaard, 2026) — the paper this repo''s Single-Turn vs Trial-Chain memory-condition manipulation (run above on the n-back task) originally comes from, there tested on a source-monitoring task across six LLMs.
    2. "Psychological Imagination Networks Show Cross-Population Centrality and Clustering Alignment in Humans That Large Language Models Fail to Replicate" (Ranjan & Odegaard, 2025) — network-science analysis of VVIQ ratings across humans and LLMs, directly relevant to the VVIQ-16 mini study run above.

    Background on the field:

    1. "Large Language Models as Psychological Simulators: A Methodological Guide" (Lin, 2025) — a methodological framework for exactly what this notebook demonstrates: LLMs standing in as simulated participants and computational models of cognition.
    2. "Can Large Language Models Replace Human Subjects? A Large-Scale Replication of Scenario-Based Experiments in Psychology and Management" (Cui, Li & Zhou, 2024) — replicates 156 published psychology/management experiments with LLM participants, at a scale far beyond this quickstart's single-trial demo.
